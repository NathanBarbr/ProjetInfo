\section{Appropriation du Sujet}
%=============================================================================

\begin{frame}{Contexte : Expertise LIRIS en Tennis de Table}
\begin{block}{Collaboration JO Paris 2024}
\begin{itemize}
    \item Le laboratoire LIRIS : expertise reconnue en analyse de performance
    \item Participation aux JO de Paris 2024 avec l'équipe de France
    \item Doctorant en appui de l'analyse vidéo
    \item Objectif : \textbf{Démocratiser ces outils vers le grand public}
\end{itemize}
\end{block}

\vspace{0.3cm}

\begin{block}{Problématique}
Comment rendre accessible au grand public des outils d'analyse vidéo professionnels utilisés en compétition olympique ?
\end{block}

\vspace{0.2cm}
\footnotesize{Source : \url{https://centralelyon.github.io/table-tennis/}}
\end{frame}

\begin{frame}{Le Défi : Recherche Multimodale}
\begin{columns}
\begin{column}{0.5\textwidth}

    \item \textcolor{green}{Sonore} : applaudissements, impacts
    \item \textcolor{orange}{Textuelle} : commentaires, métadonnées

\section{Objectifs du Projet}
%=============================================================================

\begin{frame}{Objectifs Principaux}
\begin{enumerate}
    \item \textbf{Accessibilité}
    \begin{itemize}
        \item Rendre les analyses professionnelles accessibles au grand public
        \item Interface intuitive et rapide (\textless 100ms par requête)
    \end{itemize}
    
    \vspace{0.3cm}
    
    \item \textbf{Performance}
    \begin{itemize}
        \item Indexation de centaines de matchs
        \item Recherche en temps réel avec filtres complexes
    \end{itemize}
    
    \vspace{0.3cm}
    
    \item \textbf{Intelligence}
    \begin{itemize}
        \item Intégration d'IA (RAG - Retrieval-Augmented Generation)
        \item Requêtes en langage naturel
        \item Analyses statistiques avancées
    \end{itemize}
\end{enumerate}
\end{frame}

\begin{frame}{Cas d'Usage Concrets}
\begin{block}{Pour les Entraîneurs}
\begin{itemize}
    \item Analyser les stratégies de service (long vs court)
    \item Identifier les points faibles des adversaires
    \item Préparer des séances d'entraînement ciblées
\end{itemize}
\end{block}

\begin{block}{Pour les Joueurs}
\begin{itemize}
    \item Revoir ses propres performances
    \item Étudier les techniques des champions
    \item Progresser via l'analyse vidéo
\end{itemize}
\end{block}

\begin{block}{Pour les Fans}
\begin{itemize}
    \item Retrouver les meilleurs points d'un match
    \item Explorer les statistiques des joueurs
    \item Comprendre les tactiques du jeu
\end{itemize}
\end{block}
\end{frame}

\section{État d'Avancement}
%=============================================================================

\begin{frame}{Fonctionnalités Implémentées}
\begin{block}{Backend (FastAPI)}
\begin{itemize}
    \item[$\checkmark$] \textbf{Streaming vidéo} : HTTP Range Requests, chunks 1MB
    \item[$\checkmark$] \textbf{Gestion clips} : Organisation par set/point, thumbnails
    \item[$\checkmark$] \textbf{Elasticsearch} : Indexation CSV, recherche full-text, filtres
    \item[$\checkmark$] \textbf{Chatbot RAG} : OpenAI GPT-3.5, contexte conversationnel
    \item[$\checkmark$] \textbf{Architecture modulaire} : 3 routers (videos, search, chat)
\end{itemize}
\end{block}

\begin{block}{Frontend (Next.js)}
\begin{itemize}
    \item[$\checkmark$] \textbf{Lecteur vidéo} : Contrôles avancés, navigation fluide
    \item[$\checkmark$] \textbf{Interface recherche} : Filtres multiples, pagination
    \item[$\checkmark$] \textbf{Navigation clips} : Groupement par set, aperçus visuels
    \item[$\checkmark$] \textbf{Widget chatbot} : Interface conversationnelle
\end{itemize}
\end{block}
\end{frame}
